# Xiaomi / Redmi / POCO: HyperOS 1–3 and MIUI 12–14 (Global/EEA)
# Fully tested on: REDMI 17 (2606FRN72Y), HyperOS 3.0.308, Android 16, November 2026.
#
# Format: LEVEL <TAB> ACTION <TAB> PACKAGE <TAB> # what it is
#   L1 = ads only, L2 = standard, L3 = maximum.
#   ACTION: uninstall = pm uninstall -k --user 0; disable = pm disable-user --user 0.
#   At level 1 EVERYTHING runs as disable (nothing is removed).
#   A package may be missing on a given model; that's normal, it's just skipped.

# --- L1: ads, analytics, automatic installs ---
L1	uninstall	com.miui.msa.global	# MSA, the ad service behind banners in every Xiaomi app
L1	uninstall	com.miui.msa	# MSA (Chinese/older builds)
L1	uninstall	com.miui.analytics	# Xiaomi analytics
L1	uninstall	android.autoinstalls.config.Xiaomi.model	# list of apps downloaded after setup
L1	uninstall	com.xiaomi.mipicks	# GetApps, Xiaomi's store; installs and pushes games by itself
L1	uninstall	com.miui.android.fashiongallery	# Wallpaper carousel: ads on the lock screen
L1	uninstall	com.mi.globalminusscreen	# App Vault: the feed left of the home screen
L1	uninstall	com.miui.cotaservice	# Carrier OTA: delivers carrier/partner apps
L1	uninstall	com.mi.appfinder	# Xiaomi search behind the home search bar: shows "recommended" apps/ads (disable is refused on HyperOS 3, remove works)
L1	uninstall	com.miui.hybrid	# Quick apps (ads, MIUI)
L1	uninstall	com.miui.hybrid.accessory	# Quick apps add-on
L1	uninstall	com.facebook.appmanager	# hidden Facebook installer
L1	uninstall	com.facebook.services	# hidden Facebook services
L1	uninstall	com.facebook.system	# hidden Facebook system module

# --- L2: ad-laden duplicates, stores, the manufacturer's games ---
L2	uninstall	com.miui.cleaner	# "Cleaner": ads and no real benefit
L2	uninstall	com.mi.globalbrowser	# Mi Browser, with ads (Chrome stays)
L2	uninstall	com.android.browser	# Mi Browser (older name)
L2	uninstall	com.miui.player	# Mi Music, with ads
L2	uninstall	com.miui.videoplayer	# Mi Video, with ads (videos still open from the Gallery)
L2	uninstall	com.xiaomi.glgm	# Xiaomi Game Center
L2	uninstall	com.mi.global.shop	# Mi Store (hardware shop)
L2	uninstall	com.mi.global.bbs	# Xiaomi community forum
L2	uninstall	com.miui.yellowpage	# Xiaomi "Yellow pages"/caller ID
L2	uninstall	com.xiaomi.smarthome	# Mi Home, smart home (NOT needed for Mi Band; that's Mi Fitness)
L2	uninstall	cn.wps.xiaomi.abroad.lite	# WPS Office lite, with ads
L2	uninstall	com.miui.thirdappassistant	# third-party app assistant
L2	uninstall	com.xiaomi.barrage	# "bullet comments" over games/videos
L2	uninstall	com.miui.bugreport	# bug reports
L2	uninstall	com.miui.miservice	# "Services & feedback"
L2	uninstall	com.mi.android.globalpersonalassistant	# older assistant feed (MIUI)

# --- L3: the phone works without them ---
L3	uninstall	com.xiaomi.discover	# built-in app updates. WARNING: the "Update" tab in Settings stops opening; firmware updates as before
L3	uninstall	com.miui.huanji	# Mi Mover: transfer from an old phone
L3	uninstall	com.xiaomi.midrop	# ShareMe (often protected → gets disabled)
L3	uninstall	com.xiaomi.scanner	# Scanner (the camera reads QR codes too; often protected → disabled)
L3	uninstall	com.miui.compass	# Compass (often protected)
L3	uninstall	com.miui.screenrecorder	# Screen recorder (often protected)
L3	uninstall	com.miui.fm	# FM radio (ask: elderly people listen to it!)
L3	uninstall	com.miui.notes	# Notes (ask)
L3	uninstall	com.xiaomi.calendar	# Xiaomi Calendar (if Google Calendar is there; ask)
L3	uninstall	com.miui.weather2	# Weather (ASK: the "clock + weather" widget and lock-screen weather use it)
L3	uninstall	com.mi.health	# Mi Fitness (ASK: needed for Mi Band)
L3	uninstall	com.miui.virtualsim	# Mi Roaming / virtual SIM
L3	uninstall	com.miui.vsimcore	# virtual SIM core
L3	uninstall	com.miui.misightservice	# diagnostics collection
L3	uninstall	com.xiaomi.aiasst.vision	# AI subtitles/"vision" (ask if the person is hard of hearing)
L3	uninstall	com.microsoftsdk.crossdeviceservicebroker	# link to Windows ("Phone Link")
L3	uninstall	com.android.egg	# Android easter egg

# --- DO NOT TOUCH on Xiaomi, although often suggested (see protected.txt) ---
# com.miui.securitycenter, com.lbe.security.miui, com.miui.guardprovider: bootloop / no installer
# com.xiaomi.xmsf: push notifications stop
# com.android.updater: firmware updates
# com.android.thememanager: wallpaper changes break without it (turn its ads off in its settings)
# com.miui.mediaeditor: the Gallery editor
